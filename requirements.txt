def find_similar_cases(amount: float, merchant: str) -> List[Dict]:
    '''Find similar fraud cases from knowledge base'''
    similar = []
    for case in FRAUD_CASES:
        # Simple similarity matching
        amount_diff = abs(case['amount'] - amount) / max(case['amount'], amount)
        if amount_diff < 0.5:  # Within 50% of amount
            similar.append(case)
    return similar
